import { useEffect, useState } from 'react'
import { db } from '../db/db'
import { describeDatabaseError } from '../lib/storageErrors'
import { primaryButton } from './styles'

// Shows the app only once the local database really works. If it cannot open,
// it explains what to do instead of showing a broken app.
export default function DatabaseGate({ children }) {
    const [state, setState] = useState({ status: 'loading', message: '' })
    const [attempt, setAttempt] = useState(0)

    useEffect(() => {
        let cancelled = false

        async function check() {
            try {
                await db.open()
                await db.settings.count() // a real read proves the database works
                if (!cancelled) setState({ status: 'ready', message: '' })
            } catch (err) {
                console.error('The local database could not be opened', err)
                if (!cancelled) setState({ status: 'error', message: describeDatabaseError(err) })
            }
        }

        check()
        return () => {
            cancelled = true
        }
    }, [attempt])

    function retry() {
        setState({ status: 'loading', message: '' })
        setAttempt((n) => n + 1)
    }

    if (state.status === 'loading') {
        return <p className="p-6 text-slate-500">Starting...</p>
    }

    if (state.status === 'error') {
        return (
            <div className="mx-auto max-w-lg space-y-3 p-6">
                <h1 className="text-xl font-bold text-red-700">Local storage problem</h1>
                <p>{state.message}</p>
                <ol className="list-decimal space-y-1 pl-5 text-sm text-slate-700">
                    <li>Close other tabs or windows of this app.</li>
                    <li>Free up space on the device.</li>
                    <li>
                        Do <strong>not</strong> clear the browser data. That would delete your saved sales
                        unless you have a backup file.
                    </li>
                    <li>Press &ldquo;Try again&rdquo;.</li>
                </ol>
                <button type="button" className={primaryButton} onClick={retry}>
                    Try again
                </button>
            </div>
        )
    }

    return children
}

import { NavLink, Outlet, useLocation } from 'react-router-dom'
import SyncProvider from '../sync/SyncProvider'
import BackupReminder from './BackupReminder'
import OnlinePill from './OnlinePill'
import StatusBar from './StatusBar'
import SyncBadge from './SyncBadge'
import ThemeToggle from './ThemeToggle'

const linkClass = ({ isActive }) =>
    `shrink-0 whitespace-nowrap rounded-lg px-3 py-1.5 text-sm font-medium ${isActive ? 'bg-blue-600 text-white' : 'text-slate-700 hover:bg-slate-200'
    }`

const LINKS = [
    ['/billing', 'Billing'],
    ['/sales', 'Sales'],
    ['/products', 'Products'],
    ['/labels', 'Labels'],
    ['/settings', 'Settings'],
    ['/sync', 'Sync'],
    ['/backup', 'Backup'],
    ['/storage', 'Storage'],
    ['/scanner', 'Scanner Test'],
]

function LayoutBody() {
    const { pathname } = useLocation()
    // The billing screen uses the whole window and never scrolls the page on a laptop.
    const wide = pathname.startsWith('/billing')

    return (
        <div className="flex h-dvh flex-col bg-slate-100 print:block print:h-auto print:bg-white">
            <header className="shrink-0 border-b border-slate-200 bg-white print:hidden">
                <div className="flex flex-wrap items-center gap-x-3 gap-y-1 px-3 py-2">
                    <h1 className="shrink-0 text-base font-bold">Clothing POS</h1>
                    <nav className="order-last flex w-full gap-1 overflow-x-auto md:order-none md:w-auto md:min-w-0 md:flex-1">
                        {LINKS.map(([to, label]) => (
                            <NavLink key={to} to={to} className={linkClass}>
                                {label}
                            </NavLink>
                        ))}
                    </nav>
                    <div className="ml-auto flex shrink-0 items-center gap-2 md:ml-0">
                        <OnlinePill />
                        <SyncBadge />
                        <ThemeToggle />
                    </div>
                </div>
            </header>
            <StatusBar showPill={false} />
            <BackupReminder />
            <main
                className={`min-h-0 flex-1 print:overflow-visible print:p-0 ${wide ? 'overflow-y-auto p-2 lg:overflow-hidden lg:p-3' : 'overflow-y-auto'
                    }`}
            >
                {wide ? (
                    <Outlet />
                ) : (
                    <div className="mx-auto max-w-5xl p-4 print:max-w-none print:p-0">
                        <Outlet />
                    </div>
                )}
            </main>
        </div>
    )
}

export default function Layout() {
    return (
        <SyncProvider>
            <LayoutBody />
        </SyncProvider>
    )
}
