export class NetworkError extends Error {
  constructor(message) {
    super(message)
    this.name = 'NetworkError'
  }
}

export class AuthError extends Error {
  constructor(message) {
    super(message)
    this.name = 'AuthError'
  }
}

export class ApiError extends Error {
  constructor(status, code, message, details) {
    super(message)
    this.name = 'ApiError'
    this.status = status
    this.code = code
    this.details = details
  }
}

export const DEFAULT_API_URL =
  import.meta.env.VITE_API_URL ?? 'http://localhost:5000/api'

// A small wrapper around fetch. It turns every failure into one of three kinds:
//   NetworkError  the server could not be reached (or did not answer in time)
//   AuthError     the login is missing, wrong or expired (HTTP 401)
//   ApiError      the server answered with another error
export function createApiClient({
  baseUrl = DEFAULT_API_URL,
  getToken = () => null,
  fetchImpl = (...args) => fetch(...args),
  timeoutMs = 20000,
} = {}) {
  async function request(method, path, body) {
    const controller = new AbortController()
    const timer = setTimeout(() => controller.abort(), timeoutMs)
    const timedOut = () => new NetworkError('The server did not answer in time')

    try {
      const token = getToken()
      let response
      try {
        response = await fetchImpl(`${baseUrl}${path}`, {
          method,
          headers: {
            'Content-Type': 'application/json',
            ...(token ? { Authorization: `Bearer ${token}` } : {}),
          },
          body: body === undefined ? undefined : JSON.stringify(body),
          signal: controller.signal,
        })
      } catch (err) {
        throw err?.name === 'AbortError'
          ? timedOut()
          : new NetworkError('Could not reach the server')
      }

      let data = null
      try {
        data = await response.json()
      } catch {
        data = null
      }
      if (controller.signal.aborted) throw timedOut()

      if (response.ok) return data
      if (response.status === 401) {
        throw new AuthError(data?.error?.message ?? 'Please sign in again')
      }
      throw new ApiError(
        response.status,
        data?.error?.code ?? 'HTTP_ERROR',
        data?.error?.message ??
          `The server returned an error (${response.status})`,
        data?.error?.details,
      )
    } finally {
      clearTimeout(timer)
    }
  }

  return {
    baseUrl,
    login: (username, password) =>
      request('POST', '/auth/login', { username, password }),
    push: (ops) => request('POST', '/sync/push', { ops }),
    status: () => request('GET', '/sync/status'),
    stockLevels: () => request('GET', '/sync/stock-levels'),
  }
}

// The "needs attention" list: things the server refused for a reason that retrying
// will not fix. They are kept here, and are never deleted from the device.

export async function recordAttention(op, result) {
  const id = `${op.kind}:${op.entityId}`
  const existing = await db.syncQueue.get(id)
  const now = new Date().toISOString()
  await db.syncQueue.put({
    id,
    kind: op.kind,
    entityId: op.entityId,
    entityTime: op.entityTime,
    label: op.label,
    code: result.code ?? 'REJECTED',
    message: result.message ?? '',
    status: 'open',
    createdAt: existing?.createdAt ?? now,
    updatedAt: now,
  })
}

export async function clearAttention(op) {
  await db.syncQueue.delete(`${op.kind}:${op.entityId}`)
}

export function listAttention() {
  return db.syncQueue.where('status').equals('open').sortBy('createdAt')
}

export async function dismissAttention(id) {
  await db.syncQueue.update(id, {
    status: 'dismissed',
    updatedAt: new Date().toISOString(),
  })
}

// Puts the item back in the queue, so the next sync sends it again.
export async function retryAttention(item) {
  if (item.kind === 'sale') {
    await db.sales.update(item.entityId, {
      syncStatus: 'pending',
      syncMessage: '',
    })
  } else if (item.kind === 'variant') {
    await forgetVariantHash(item.entityId)
  } else {
    await rewindCursor(item.kind, item.entityTime)
  }
  await db.syncQueue.delete(item.id)
}

export async function retryAllAttention() {
  for (const item of await listAttention()) await retryAttention(item)
}

// How long to wait before the automatic retry, after this many failures in a row:
// 30 seconds, 1 minute, 2 minutes... never more than 15 minutes.
export function backoffMs(attempt) {
  if (attempt <= 0) return 0
  return Math.min(15 * 60 * 1000, 30_000 * 2 ** (attempt - 1))
}

// Sales dated more than 10 minutes in the future are refused by the server, so a
// wrong clock on this device would block them. Warn well before that.
export const CLOCK_WARNING_MS = 2 * 60 * 1000

// skewMs > 0 means this device's clock is AHEAD of the server's.
export async function checkClock(api, now = () => Date.now()) {
  const before = now()
  const { serverTime } = await api.status()
  const after = now()
  const skewMs = Math.round((before + after) / 2 - Date.parse(serverTime))
  return {
    skewMs,
    ok: Math.abs(skewMs) <= CLOCK_WARNING_MS,
    checkedAt: new Date(after).toISOString(),
  }
}

import { db } from '../db/db'
import { getCursor, getVariantHashes } from './cursors'
import { movementOp, productOp, saleOp, variantOp } from './ops'

const byKey = (a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : 0)

async function pendingCatalogAndStock() {
  const productCursor = await getCursor('product')
  const movementCursor = await getCursor('movement')
  const hashes = await getVariantHashes()

  const products = (await db.products.toArray())
    .map(productOp)
    .filter((op) => op.key > productCursor)
    .sort(byKey)

  // Only a change to an item's catalogue fields makes it pending. A stock change does not.
  const variants = (await db.variants.toArray())
    .map(variantOp)
    .filter((op) => hashes[op.entityId] !== op.fingerprint)
    .sort(byKey)

  // Sale movements are not sent: the server makes its own when it saves the sale.
  const movements = (await db.stockMovements.toArray())
    .filter((m) => m.type !== 'sale')
    .map(movementOp)
    .filter((op) => op.key > movementCursor)
    .sort(byKey)

  return { products, variants, movements }
}

// Everything waiting to be sent, in the order the server needs it:
// products, then items, then stock changes, then sales. Oldest first within each.
export async function collectPending() {
  const { products, variants, movements } = await pendingCatalogAndStock()

  const sales = await db.sales.where('syncStatus').equals('pending').toArray()
  const saleKey = (sale) => `${sale.createdAt}|${sale.id}`
  sales.sort((a, b) =>
    saleKey(a) < saleKey(b) ? -1 : saleKey(a) > saleKey(b) ? 1 : 0,
  )

  const saleOps = []
  for (const sale of sales) {
    const items = await db.saleItems.where('saleId').equals(sale.id).toArray()
    saleOps.push(saleOp(sale, items))
  }

  return [...products, ...variants, ...movements, ...saleOps]
}

export function chunk(list, size) {
  const batches = []
  for (let i = 0; i < list.length; i += size)
    batches.push(list.slice(i, i + size))
  return batches
}

export async function countPending() {
  const { products, variants, movements } = await pendingCatalogAndStock()
  const sales = await db.sales.where('syncStatus').equals('pending').count()
  const attention = await db.syncQueue.where('status').equals('open').count()
  return {
    products: products.length,
    variants: variants.length,
    movements: movements.length,
    sales,
    total: products.length + variants.length + movements.length + sales,
    attention,
  }
}

import { db } from '../db/db'

// Products and stock changes are tracked with a cursor: "everything after this point
// has been sent". Items are tracked by a fingerprint of their catalogue fields.
const CURSOR_KEYS = {
  product: 'sync.productCursor',
  movement: 'sync.movementCursor',
}
const HASHES_KEY = 'sync.variantHashes'

export async function getCursor(kind) {
  return (await db.settings.get(CURSOR_KEYS[kind]))?.value ?? ''
}

export async function setCursor(kind, value) {
  await db.settings.put({ key: CURSOR_KEYS[kind], value })
}

// Moves the cursor back so a rejected item is picked up again.
export async function rewindCursor(kind, entityTime) {
  const target = `${entityTime}|`
  const current = await getCursor(kind)
  if (target < current) await setCursor(kind, target)
}

export async function getVariantHashes() {
  return (await db.settings.get(HASHES_KEY))?.value ?? {}
}

export async function markVariantsSynced(ops) {
  if (ops.length === 0) return
  const hashes = { ...(await getVariantHashes()) }
  for (const op of ops) hashes[op.entityId] = op.fingerprint
  await db.settings.put({ key: HASHES_KEY, value: hashes })
}

export async function forgetVariantHash(variantId) {
  const hashes = { ...(await getVariantHashes()) }
  delete hashes[variantId]
  await db.settings.put({ key: HASHES_KEY, value: hashes })
}

import { db } from '../db/db'
import { AuthError, NetworkError } from './apiClient'
import { clearAttention, recordAttention } from './attention'
import { chunk, collectPending, countPending } from './collect'
import { markVariantsSynced, setCursor } from './cursors'
import { toRequestOp } from './ops'

const BATCH_SIZE = 20

let running = null

// Sends everything that is waiting. Only one run happens at a time: a second call
// while one is running simply waits for, and returns, the same run.
//
// outcome: 'ok' | 'offline' | 'auth' | 'server-error'
export function runSync(options) {
  if (running) return running
  running = doSync(options).finally(() => {
    running = null
  })
  return running
}

async function doSync({ api, batchSize = BATCH_SIZE }) {
  const summary = {
    outcome: 'ok',
    sent: 0,
    applied: 0,
    duplicate: 0,
    rejected: 0,
    error: 0,
    remaining: 0,
    message: '',
  }

  const ops = await collectPending()

  for (const batch of chunk(ops, batchSize)) {
    let response
    try {
      response = await api.push(batch.map(toRequestOp))
    } catch (err) {
      if (err instanceof NetworkError) {
        summary.outcome = 'offline'
        summary.message = err.message
      } else if (err instanceof AuthError) {
        summary.outcome = 'auth'
        summary.message = 'Your sign-in is no longer valid.'
      } else {
        summary.outcome = 'server-error'
        summary.message = err?.message ?? 'The server returned an error.'
      }
      break
    }

    const results = response?.results
    if (!Array.isArray(results) || results.length !== batch.length) {
      summary.outcome = 'server-error'
      summary.message = 'The server sent an unexpected answer.'
      break
    }

    summary.sent += batch.length
    const temporary = await applyBatchResults(batch, results, summary)
    if (temporary) {
      summary.outcome = 'server-error'
      summary.message =
        'The server had a temporary problem. It will be tried again.'
      break
    }
  }

  summary.remaining = (await countPending()).total
  await saveLastRun(summary)
  return summary
}

// Records what the server said about each operation. Returns true if any of them
// hit a temporary error (those are tried again later).
async function applyBatchResults(batch, results, summary) {
  let temporary = false

  for (let i = 0; i < batch.length; i += 1) {
    const op = batch[i]
    const result = results[i]

    if (result?.status === 'applied' || result?.status === 'duplicate') {
      summary[result.status] += 1
      if (op.kind === 'sale') {
        await db.sales.update(op.entityId, {
          syncStatus: 'synced',
          syncedAt: new Date().toISOString(),
          syncFlags: result.flags ?? [],
          syncMessage: '',
        })
      }
      await clearAttention(op)
    } else if (result?.status === 'rejected') {
      summary.rejected += 1
      await recordAttention(op, result)
      if (op.kind === 'sale') {
        await db.sales.update(op.entityId, {
          syncStatus: 'review',
          syncMessage: result.message ?? '',
        })
      }
    } else {
      summary.error += 1
      temporary = true
    }
  }

  // Only move forward when the whole batch is settled. Otherwise the batch is sent
  // again next time, which is safe because the server ignores what it already has.
  if (!temporary) await advance(batch)
  return temporary
}

async function advance(batch) {
  const lastKey = {}
  for (const op of batch) {
    if (op.kind === 'product' || op.kind === 'movement')
      lastKey[op.kind] = op.key
  }
  for (const [kind, key] of Object.entries(lastKey)) await setCursor(kind, key)
  await markVariantsSynced(batch.filter((op) => op.kind === 'variant'))
}

async function saveLastRun(summary) {
  const at = new Date().toISOString()
  await db.settings.put({
    key: 'sync.lastRun',
    value: {
      at,
      outcome: summary.outcome,
      message: summary.message,
      applied: summary.applied,
      duplicate: summary.duplicate,
      rejected: summary.rejected,
      remaining: summary.remaining,
    },
  })
  if (summary.outcome === 'ok' && summary.sent > 0) {
    await db.settings.put({ key: 'sync.lastSuccessAt', value: at })
  }
}

// Turns rows from the device's database into the operations the server understands.
// Each operation also carries a few fields for the engine's own use (kind, key, ...);
// toRequestOp() removes them before anything is sent.

// A short fingerprint of some text. It is used only to notice that an item's
// catalogue fields changed. It is not security.
export function fingerprint(text) {
  let a = 0x811c9dc5
  let b = 0x84222325
  for (let i = 0; i < text.length; i += 1) {
    const code = text.charCodeAt(i)
    a = Math.imul(a ^ code, 0x01000193) >>> 0
    b = Math.imul(b ^ code, 0x1b873593) >>> 0
  }
  return a.toString(16).padStart(8, '0') + b.toString(16).padStart(8, '0')
}

const VARIANT_FIELDS = [
  'productId',
  'sku',
  'barcode',
  'size',
  'costPaise',
  'mrpPaise',
  'sellingPricePaise',
  'taxRatePercent',
  'priceIncludesTax',
  'reorderThreshold',
  'status',
]

export function variantOp(variant) {
  const v = {
    ...variant,
    taxRatePercent: variant.taxRatePercent ?? 0,
    priceIncludesTax: variant.priceIncludesTax ?? true,
    reorderThreshold: variant.reorderThreshold ?? 0,
  }
  const data = { id: v.id }
  for (const field of VARIANT_FIELDS) data[field] = v[field]

  return {
    type: 'variant.upsert',
    data,
    kind: 'variant',
    entityId: v.id,
    entityTime: v.createdAt,
    key: `${v.createdAt}|${v.id}`,
    label: v.sku,
    fingerprint: fingerprint(
      JSON.stringify(VARIANT_FIELDS.map((field) => v[field])),
    ),
  }
}

export function movementOp(movement) {
  const change = movement.quantityChange
  return {
    type: 'stock.movement',
    data: {
      id: movement.id,
      variantId: movement.variantId,
      type: movement.type,
      quantityChange: change,
      reason: String(movement.reason ?? '').slice(0, 200),
      createdAt: movement.createdAt,
    },
    kind: 'movement',
    entityId: movement.id,
    entityTime: movement.createdAt,
    key: `${movement.createdAt}|${movement.id}`,
    label: `Stock ${movement.type} ${change > 0 ? '+' : ''}${change}`,
  }
}

const discountOf = (discount) => ({
  type: discount.type,
  value: discount.value,
})

// The server recalculates every total itself. Only what the customer saw (the unit
// prices and discounts), which items had discounts switched off, and the total the
// device calculated (to compare) are sent.
export function saleOp(sale, items) {
  const lines = [...items]
    .sort((a, b) => a.lineNumber - b.lineNumber)
    .map((item) => ({
      variantId: item.variantId,
      quantity: item.quantity,
      unitPricePaise: item.unitPricePaise,
      ...(item.discount ? { discount: discountOf(item.discount) } : {}),
      ...(item.discountExcluded ? { noDiscount: true } : {}),
    }))

  const payment = { method: sale.payment.method }
  if (sale.payment.method === 'cash')
    payment.amountTenderedPaise = sale.payment.amountTenderedPaise
  if (sale.payment.reference) payment.reference = sale.payment.reference

  return {
    type: 'sale.create',
    data: {
      id: sale.id,
      invoiceNumber: sale.invoiceNumber,
      soldAt: sale.createdAt,
      offline: true,
      lines,
      ...(sale.billDiscount
        ? { billDiscount: discountOf(sale.billDiscount) }
        : {}),
      payment,
      claimedTotalPaise: sale.totalPaise,
    },
    kind: 'sale',
    entityId: sale.id,
    entityTime: sale.createdAt,
    key: `${sale.createdAt}|${sale.id}`,
    label: sale.invoiceNumber,
  }
}

// The sign-in is kept in localStorage, NOT in the database. That keeps the login
// token out of backup files.
export const SESSION_KEY = 'clothing-pos-session'

const storage = () => globalThis.localStorage

export function loadSession() {
  try {
    const raw = storage()?.getItem(SESSION_KEY)
    if (!raw) return null
    const session = JSON.parse(raw)
    return session && typeof session.token === 'string' ? session : null
  } catch {
    return null
  }
}

export function saveSession(session) {
  try {
    storage()?.setItem(SESSION_KEY, JSON.stringify(session))
    return true
  } catch {
    return false
  }
}

export function clearSession() {
  try {
    storage()?.removeItem(SESSION_KEY)
  } catch {
    // nothing to do
  }
}

// Reads the expiry time (in milliseconds) from a login token, or null.
export function tokenExpiry(token) {
  try {
    const payload = token.split('.')[1]
    const json = atob(payload.replace(/-/g, '+').replace(/_/g, '/'))
    const { exp } = JSON.parse(json)
    return typeof exp === 'number' ? exp * 1000 : null
  } catch {
    return null
  }
}

export function isExpired(session, now = Date.now()) {
  const expiry = tokenExpiry(session?.token ?? '')
  return expiry !== null && expiry <= now
}

// Compares this device's stock with the server's, item by item. Differences are
// reported, never "fixed" automatically.
export async function compareStock(api) {
  const { items } = await api.stockLevels()
  const server = new Map(items.map((item) => [item.variantId, item]))
  const local = await db.variants.toArray()

  const mismatches = []
  for (const variant of local) {
    const remote = server.get(variant.id)
    if (!remote) {
      mismatches.push({
        variantId: variant.id,
        sku: variant.sku,
        local: variant.currentStock,
        server: null,
      })
    } else if (remote.currentStock !== variant.currentStock) {
      mismatches.push({
        variantId: variant.id,
        sku: variant.sku,
        local: variant.currentStock,
        server: remote.currentStock,
      })
    }
  }

  const localIds = new Set(local.map((variant) => variant.id))
  const onlyOnServer = items
    .filter((item) => !localIds.has(item.variantId))
    .map((item) => ({
      variantId: item.variantId,
      sku: item.sku,
      server: item.currentStock,
    }))

  return { checked: local.length, mismatches, onlyOnServer }
}

import { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState } from 'react'
import { useLiveQuery } from 'dexie-react-hooks'
import { db } from '../db/db'
import { useOnlineStatus } from '../pwa/usePwa'
import { createApiClient } from './apiClient'
import { backoffMs } from './backoff'
import { checkClock } from './clock'
import { countPending } from './collect'
import { runSync } from './engine'
import { clearSession, isExpired, loadSession, saveSession } from './session'

export const SyncContext = createContext(null)

export function useSync() {
    const value = useContext(SyncContext)
    if (!value) throw new Error('useSync must be used inside SyncProvider')
    return value
}

const EMPTY_COUNTS = {
    products: 0,
    variants: 0,
    movements: 0,
    sales: 0,
    total: 0,
    attention: 0,
}

// `api`, `autoSync`, `autoDelayMs` and `intervalMs` can be replaced in tests.
export default function SyncProvider({
    children,
    api: apiProp,
    autoSync = true,
    autoDelayMs = 2000,
    intervalMs = 60_000,
}) {
    const [session, setSession] = useState(() => loadSession())
    const sessionRef = useRef(session)
    const online = useOnlineStatus()
    const [syncing, setSyncing] = useState(false)
    const [clock, setClock] = useState(null)
    const [notice, setNotice] = useState('')
    const busy = useRef(false)
    const retry = useRef({ attempt: 0, at: 0 })

    const api = useMemo(
        () => apiProp ?? createApiClient({ getToken: () => sessionRef.current?.token ?? null }),
        [apiProp],
    )

    const counts = useLiveQuery(() => countPending(), []) ?? EMPTY_COUNTS
    const lastRun = useLiveQuery(async () => (await db.settings.get('sync.lastRun'))?.value ?? null, [])
    const lastSuccessAt = useLiveQuery(
        async () => (await db.settings.get('sync.lastSuccessAt'))?.value ?? null,
        [],
    )

    const updateSession = useCallback((next) => {
        sessionRef.current = next
        setSession(next)
    }, [])

    const endSession = useCallback(
        (message) => {
            clearSession()
            updateSession(null)
            setNotice(message)
        },
        [updateSession],
    )

    const signedIn = Boolean(session) && !isExpired(session)

    const run = useCallback(
        async ({ manual = false } = {}) => {
            const current = sessionRef.current
            if (busy.current || !current) return null
            if (isExpired(current)) {
                endSession('Your sign-in has expired. Please sign in again to sync.')
                return null
            }
            if (!manual && Date.now() < retry.current.at) return null

            busy.current = true
            setSyncing(true)
            try {
                let summary
                try {
                    summary = await runSync({ api })
                } catch (err) {
                    console.error('Sync stopped because of a problem on this device', err)
                    summary = { outcome: 'server-error', sent: 0, message: 'Sync stopped because of a problem on this device.' }
                }

                if (summary.outcome === 'ok') {
                    retry.current = { attempt: 0, at: 0 }
                    setNotice('')
                    if (summary.sent > 0 || manual) {
                        try {
                            setClock(await checkClock(api))
                        } catch {
                            // the clock check is optional
                        }
                    }
                } else if (summary.outcome === 'auth') {
                    endSession('Your sign-in is no longer valid. Please sign in again to sync.')
                } else {
                    const attempt = retry.current.attempt + 1
                    retry.current = { attempt, at: Date.now() + backoffMs(attempt) }
                    setNotice(summary.message)
                }
                return summary
            } finally {
                busy.current = false
                setSyncing(false)
            }
        },
        [api, endSession],
    )

    // Sync shortly after something becomes pending, or when we come back online.
    useEffect(() => {
        if (!autoSync || !session || !online || counts.total === 0) return undefined
        const timer = setTimeout(() => {
            run()
        }, autoDelayMs)
        return () => clearTimeout(timer)
    }, [autoSync, session, online, counts.total, run, autoDelayMs])

    // A safety net: check again every minute (retries wait for their back-off time).
    useEffect(() => {
        if (!autoSync || !session) return undefined
        const timer = setInterval(() => {
            run()
        }, intervalMs)
        return () => clearInterval(timer)
    }, [autoSync, session, run, intervalMs])

    const signIn = useCallback(
        async (username, password) => {
            const { token, user } = await api.login(username, password)
            const next = { token, user, signedInAt: new Date().toISOString() }
            saveSession(next)
            updateSession(next)
            retry.current = { attempt: 0, at: 0 }
            setNotice('')
            return next
        },
        [api, updateSession],
    )

    const signOut = useCallback(() => {
        clearSession()
        updateSession(null)
        setNotice('')
    }, [updateSession])

    const value = useMemo(
        () => ({
            session,
            signedIn,
            online,
            syncing,
            counts,
            lastRun: lastRun ?? null,
            lastSuccessAt: lastSuccessAt ?? null,
            clock,
            notice,
            api,
            signIn,
            signOut,
            syncNow: () => run({ manual: true }),
        }),
        [session, signedIn, online, syncing, counts, lastRun, lastSuccessAt, clock, notice, api, signIn, signOut, run],
    )

    return <SyncContext.Provider value={value}>{children}</SyncContext.Provider>
}

// All amounts are integer paise. Percentages and tax rates are converted to
// basis points (1% = 100 bp) so no floating-point money is ever used.

// Integer division rounded half-up. Only for non-negative numbers.
export function divRound(numerator, denominator) {
  return Math.floor((2 * numerator + denominator) / (2 * denominator))
}

const toBasisPoints = (percent) => Math.round(percent * 100)

// Splits `totalPaise` across `weights` in proportion, with no paisa lost:
// every share is rounded down, then the leftover paise go to the largest
// remainders. The shares always add up to exactly `totalPaise`.
export function allocateProportionally(totalPaise, weights) {
  const sum = weights.reduce((a, b) => a + b, 0)
  if (totalPaise <= 0 || sum <= 0) return weights.map(() => 0)

  const total = BigInt(totalPaise)
  const weightSum = BigInt(sum)
  const shares = weights.map((w) => Number((total * BigInt(w)) / weightSum))
  const remainders = weights.map((w, index) => ({
    index,
    remainder: (total * BigInt(w)) % weightSum,
  }))

  const leftover = totalPaise - shares.reduce((a, b) => a + b, 0)
  remainders.sort((a, b) => {
    if (a.remainder === b.remainder) return a.index - b.index
    return a.remainder > b.remainder ? -1 : 1
  })
  for (let k = 0; k < leftover; k += 1) shares[remainders[k].index] += 1
  return shares
}

// A discount is { type: 'percent', value: 10 } or { type: 'amount', value: 10000 }
// (amount is in paise). The result can never exceed the amount it applies to.
export function discountAmount(discount, basePaise) {
  if (!discount || basePaise <= 0) return 0
  if (discount.type === 'percent') {
    return divRound(basePaise * toBasisPoints(discount.value), 10000)
  }
  return Math.min(discount.value, basePaise)
}

export function calculateTotals(cart, { roundToRupee = true } = {}) {
  const base = cart.lines.map((line) => {
    const grossPaise = line.unitPricePaise * line.quantity
    const lineDiscountPaise = discountAmount(line.discount, grossPaise)
    return {
      line,
      grossPaise,
      lineDiscountPaise,
      afterItemPaise: grossPaise - lineDiscountPaise,
    }
  })

  const afterItemTotal = base.reduce((sum, b) => sum + b.afterItemPaise, 0)
  const billDiscountPaise = discountAmount(cart.billDiscount, afterItemTotal)
  const shares = allocateProportionally(
    billDiscountPaise,
    base.map((b) => b.afterItemPaise),
  )

  const lines = base.map((b, i) => {
    const netPaise = b.afterItemPaise - shares[i]
    const bp = toBasisPoints(b.line.taxRatePercent ?? 0)
    const taxIncluded = b.line.priceIncludesTax ?? true

    let taxPaise
    let taxableValuePaise
    let lineTotalPaise
    if (taxIncluded) {
      taxPaise = divRound(netPaise * bp, 10000 + bp)
      taxableValuePaise = netPaise - taxPaise
      lineTotalPaise = netPaise
    } else {
      taxPaise = divRound(netPaise * bp, 10000)
      taxableValuePaise = netPaise
      lineTotalPaise = netPaise + taxPaise
    }

    return {
      variantId: b.line.variantId,
      quantity: b.line.quantity,
      unitPricePaise: b.line.unitPricePaise,
      grossPaise: b.grossPaise,
      lineDiscountPaise: b.lineDiscountPaise,
      billDiscountSharePaise: shares[i],
      netPaise,
      taxableValuePaise,
      taxPaise,
      taxIncluded,
      lineTotalPaise,
    }
  })

  const sum = (key) => lines.reduce((total, l) => total + l[key], 0)
  const taxPaise = sum('taxPaise')
  const taxIncludedPaise = lines
    .filter((l) => l.taxIncluded)
    .reduce((total, l) => total + l.taxPaise, 0)
  const exactTotalPaise = sum('lineTotalPaise')
  const totalPaise = roundToRupee
    ? divRound(exactTotalPaise, 100) * 100
    : exactTotalPaise

  return {
    lines,
    itemCount: lines.reduce((total, l) => total + l.quantity, 0),
    subtotalPaise: sum('grossPaise'),
    itemDiscountPaise: sum('lineDiscountPaise'),
    billDiscountPaise,
    discountPaise: sum('lineDiscountPaise') + billDiscountPaise,
    taxableValuePaise: sum('taxableValuePaise'),
    taxPaise,
    taxIncludedPaise,
    taxAddedPaise: taxPaise - taxIncludedPaise,
    exactTotalPaise,
    roundOffPaise: totalPaise - exactTotalPaise,
    totalPaise,
  }
}

import { createLimiter } from "../http/rateLimiter.js";
import { parse } from "../http/validate.js";
import { User, publicUser } from "../models/User.js";
import { audit } from "../services/audit.js";

const loginSchema = z
  .object({
    username: z.string().trim().min(1).max(60),
    password: z.string().min(1).max(200),
  })
  .strict();

const changePasswordSchema = z
  .object({
    currentPassword: z.string().min(1).max(200),
    newPassword: passwordSchema,
  })
  .strict();

export function authRouter(config) {
  const router = Router();
  const loginLimiter = createLimiter(
    config.loginRateLimit,
    "Too many login attempts. Please try again later.",
  );
  const authenticate = requireAuth(config);
  let dummyHash = null;

  // Spends the same time a real password check would, so the response time does
  // not reveal whether a username exists.
  async function burnTime(password) {
    dummyHash ??= await hashPassword(randomUUID(), config.bcryptCost);
    await verifyPassword(password, dummyHash);
  }

  router.post(
    "/login",
    loginLimiter,
    asyncHandler(async (req, res) => {
      const { username, password } = parse(loginSchema, req.body);
      const name = username.toLowerCase();
      const user = await User.findOne({ username: name }).select(
        "+passwordHash",
      );
      const now = new Date();

      const reject = async (reason) => {
        await audit(req, {
          actor: null,
          action: "login.failed",
          entity: "user",
          entityId: user?._id ?? "",
          details: { username: name, reason },
        });
        throw unauthorized("Invalid username or password");
      };

      if (!user || !user.active) {
        await burnTime(password);
        return reject(user ? "account disabled" : "unknown user");
      }

      if (user.lockedUntil && user.lockedUntil > now) {
        const minutes = Math.max(
          1,
          Math.ceil((user.lockedUntil - now) / 60000),
        );
        throw new AppError(
          429,
          "ACCOUNT_LOCKED",
          `Too many failed attempts. Try again in ${minutes} minute${minutes === 1 ? "" : "s"}.`,
        );
      }

      if (!(await verifyPassword(password, user.passwordHash))) {
        user.failedLogins += 1;
        if (user.failedLogins >= config.lockout.maxFailures) {
          user.lockedUntil = new Date(
            now.getTime() + config.lockout.minutes * 60000,
          );
          user.failedLogins = 0;
          await audit(req, {
            actor: user,
            action: "account.locked",
            entity: "user",
            entityId: user._id,
          });
        }
        await user.save();
        return reject("wrong password");
      }

      user.failedLogins = 0;
      user.lockedUntil = null;
      user.lastLoginAt = now;
      await user.save();

      await audit(req, {
        actor: user,
        action: "login.succeeded",
        entity: "user",
        entityId: user._id,
      });
      res.json({ token: signToken(user, config), user: publicUser(user) });
    }),
  );

  router.get("/me", authenticate, (req, res) => {
    res.json({ user: publicUser(req.user) });
  });

  router.post(
    "/change-password",
    authenticate,
    asyncHandler(async (req, res) => {
      const { currentPassword, newPassword } = parse(
        changePasswordSchema,
        req.body,
      );
      const user = await User.findById(req.user._id).select("+passwordHash");

      if (!(await verifyPassword(currentPassword, user.passwordHash))) {
        throw new AppError(
          400,
          "WRONG_PASSWORD",
          "Your current password is not correct",
        );
      }
      if (newPassword === currentPassword) {
        throw new AppError(
          400,
          "VALIDATION_ERROR",
          "The new password must be different",
          [{ field: "newPassword", message: "Choose a different password" }],
        );
      }

      user.passwordHash = await hashPassword(newPassword, config.bcryptCost);
      user.tokenVersion += 1; // logs out every other session
      await user.save();

      await audit(req, {
        action: "password.changed",
        entity: "user",
        entityId: user._id,
      });
      res.json({ token: signToken(user, config), user: publicUser(user) });
    }),
  );

  return router;
}

import { createHash } from "node:crypto";
import { runInTransaction } from "../db/transaction.js";
import { calculateTotals } from "../domain/calc.js";
import { AppError, conflict } from "../errors.js";
import { nextSequence } from "../models/Counter.js";
import { Product } from "../models/Product.js";
import { Sale } from "../models/Sale.js";
import { getShopSettings } from "../models/Settings.js";
import { Variant } from "../models/Variant.js";
import { applyStockChange } from "./stock.js";

const FUTURE_TOLERANCE_MS = 10 * 60 * 1000;

const invalid = (field, message) =>
  new AppError(400, "VALIDATION_ERROR", "Some fields are invalid", [
    { field, message },
  ]);

// A fingerprint of what the sale says. The same id with different content is refused.
function fingerprintOf(data) {
  const lines = data.lines
    .map((l) => [l.variantId, l.quantity, l.unitPricePaise, l.discount ?? null])
    .sort((a, b) => (a[0] < b[0] ? -1 : a[0] > b[0] ? 1 : 0));
  const canonical = {
    lines,
    billDiscount: data.billDiscount ?? null,
    payment: [data.payment.method, data.payment.amountTenderedPaise ?? null],
    invoiceNumber: data.invoiceNumber ?? null,
  };
  return createHash("sha256").update(JSON.stringify(canonical)).digest("hex");
}

function existingResult(existing, fingerprint) {
  if (existing.requestHash !== fingerprint) {
    throw conflict(
      "IDEMPOTENCY_KEY_REUSED",
      "A different sale was already saved with this id",
    );
  }
  return { sale: existing, duplicate: true };
}

// Saves a sale. Returns { sale, duplicate }.
//   online sale  : a broken shop rule REFUSES the sale (422/409)
//   offline sale : the sale already happened, so it is saved and FLAGGED instead
export async function createSale({ data, user }) {
  const settings = await getShopSettings();
  const fingerprint = fingerprintOf(data);
  const offline = data.offline === true;

  try {
    return await runInTransaction((session) =>
      saveSale({ data, user, settings, fingerprint, offline, session }),
    );
  } catch (err) {
    if (err?.code === 11000) {
      const fields = Object.keys(err.keyPattern ?? err.keyValue ?? {});
      if (fields.includes("_id")) {
        // Another request saved this same sale at the same moment.
        const existing = await Sale.findById(data.id);
        if (existing) return existingResult(existing, fingerprint);
      }
      if (fields.includes("invoiceNumber")) {
        throw conflict(
          "INVOICE_NUMBER_TAKEN",
          "That invoice number is already used by another sale",
        );
      }
    }
    throw err;
  }
}

async function saveSale({
  data,
  user,
  settings,
  fingerprint,
  offline,
  session,
}) {
  // 1. Already saved? Then this is a repeat of the same request.
  const existing = await Sale.findById(data.id).session(session);
  if (existing) return existingResult(existing, fingerprint);

  // 2. The sale time must not be in the future.
  const now = new Date();
  const soldAt = data.soldAt ? new Date(data.soldAt) : now;
  if (soldAt.getTime() > now.getTime() + FUTURE_TOLERANCE_MS) {
    throw invalid("soldAt", "The sale time is in the future");
  }

  // 3. Load the items and their products.
  const variants = await Variant.find({
    _id: { $in: data.lines.map((l) => l.variantId) },
  }).session(session);
  const variantById = new Map(variants.map((v) => [v._id, v]));
  const products = await Product.find({
    _id: { $in: [...new Set(variants.map((v) => v.productId))] },
  }).session(session);
  const productById = new Map(products.map((p) => [p._id, p]));

  // 4. Check every line against the catalogue. Problems are collected, not thrown.
  const violations = [];
  const lineViews = data.lines.map((line) => {
    const variant = variantById.get(line.variantId);
    if (!variant) {
      throw new AppError(
        422,
        "UNKNOWN_VARIANT",
        `Item ${line.variantId} does not exist`,
        [{ variantId: line.variantId }],
      );
    }
    const product = productById.get(variant.productId);
    const label = `${product?.name ?? "Item"} (size ${variant.size})`;

    if (variant.status !== "active" || product?.status !== "active") {
      violations.push({
        code: "ITEM_ARCHIVED",
        variantId: variant._id,
        message: `${label} is archived`,
      });
    }
    if (line.unitPricePaise !== variant.sellingPricePaise) {
      violations.push({
        code: "PRICE_DIFFERS_FROM_CATALOG",
        variantId: variant._id,
        message: `${label} was sold at ${line.unitPricePaise} paise, the catalogue price is ${variant.sellingPricePaise}`,
      });
    }
    if (line.unitPricePaise > variant.mrpPaise) {
      violations.push({
        code: "PRICE_ABOVE_MRP",
        variantId: variant._id,
        message: `${label} was sold above its MRP`,
      });
    }
    return { line, variant, product };
  });

  // 5. The server's own calculation. Tax settings come from the catalogue.
  const options = { roundToRupee: settings.roundToRupee };
  const cart = {
    lines: lineViews.map(({ line, variant }) => ({
      variantId: variant._id,
      unitPricePaise: line.unitPricePaise,
      quantity: line.quantity,
      taxRatePercent: variant.taxRatePercent,
      priceIncludesTax: variant.priceIncludesTax,
      discount: line.discount ?? null,
    })),
    billDiscount: data.billDiscount ?? null,
  };

  // Impossible discounts are refused in every mode.
  for (const { line } of lineViews) {
    if (
      line.discount?.type === "amount" &&
      line.discount.value > line.unitPricePaise * line.quantity
    ) {
      throw invalid("lines", "A discount cannot be more than the item total");
    }
  }
  const beforeBillDiscount = calculateTotals(
    { ...cart, billDiscount: null },
    options,
  );
  if (
    cart.billDiscount?.type === "amount" &&
    cart.billDiscount.value >
      beforeBillDiscount.subtotalPaise - beforeBillDiscount.itemDiscountPaise
  ) {
    throw invalid(
      "billDiscount",
      "A discount cannot be more than the bill total",
    );
  }

  const totals = calculateTotals(cart, options);

  // 6. Cashier discount limit (admins are not limited). A little slack for rounding.
  if (user.role === "cashier") {
    const limitBp = Math.round(settings.maxCashierDiscountPercent * 100);
    const allowed =
      Math.ceil((totals.subtotalPaise * limitBp) / 10000) +
      cart.lines.length +
      1;
    if (totals.discountPaise > allowed) {
      violations.push({
        code: "DISCOUNT_OVER_LIMIT",
        message: `The discount is more than the ${settings.maxCashierDiscountPercent}% a cashier may give`,
      });
    }
  }

  // 7. The total the client claims must match the server's total.
  if (
    data.claimedTotalPaise !== undefined &&
    data.claimedTotalPaise !== totals.totalPaise
  ) {
    violations.push({
      code: "TOTAL_MISMATCH",
      message: `The total sent (${data.claimedTotalPaise}) differs from the calculated total (${totals.totalPaise})`,
    });
  }

  // 8. Payment.
  const isCash = data.payment.method === "cash";
  const tendered = isCash
    ? (data.payment.amountTenderedPaise ?? totals.totalPaise)
    : totals.totalPaise;
  let changePaise = 0;
  if (tendered < totals.totalPaise) {
    if (!offline)
      throw new AppError(
        400,
        "INSUFFICIENT_PAYMENT",
        "Cash received is less than the total",
      );
    violations.push({
      code: "PAYMENT_SHORT",
      message: `The amount received (${tendered}) is less than the total (${totals.totalPaise})`,
    });
  } else {
    changePaise = tendered - totals.totalPaise;
  }

  // 9. An online sale that breaks any rule is refused. An offline one is flagged.
  if (violations.length > 0 && !offline) {
    throw new AppError(
      422,
      "SALE_NOT_ALLOWED",
      "This sale breaks a shop rule",
      violations,
    );
  }
  const flags = [...violations];

  // 10. Invoice number: keep the device's own, or allocate one from the server series.
  let invoiceNumber = data.invoiceNumber;
  if (!invoiceNumber) {
    const sequence = await nextSequence("invoice", session);
    invoiceNumber = `${settings.invoicePrefix}-S${String(sequence).padStart(6, "0")}`;
  }

  // 11. Stock. A fixed order keeps concurrent sales from blocking each other.
  const ordered = [...lineViews].sort((a, b) =>
    a.variant._id < b.variant._id ? -1 : 1,
  );
  for (const { line, variant } of ordered) {
    const { variant: after } = await applyStockChange({
      variantId: variant._id,
      quantityChange: -line.quantity,
      type: "sale",
      referenceId: data.id,
      userId: user._id,
      allowNegative: offline || settings.allowNegativeStock,
      session,
    });
    if (offline && !settings.allowNegativeStock && after.currentStock < 0) {
      flags.push({
        code: "STOCK_SHORTFALL",
        variantId: variant._id,
        message: `${variant.sku} is now ${-after.currentStock} below zero`,
      });
    }
  }

  // 12. Save the sale with a snapshot of every line.
  const items = lineViews.map(({ line, variant, product }, index) => {
    const t = totals.lines[index];
    return {
      lineNumber: index + 1,
      variantId: variant._id,
      productId: variant.productId,
      productName: product?.name ?? "",
      sku: variant.sku,
      barcode: variant.barcode,
      size: variant.size,
      quantity: line.quantity,
      unitPricePaise: line.unitPricePaise,
      mrpPaise: variant.mrpPaise,
      costPaise: variant.costPaise,
      taxRatePercent: variant.taxRatePercent,
      priceIncludesTax: variant.priceIncludesTax,
      discount: line.discount ?? null,
      grossPaise: t.grossPaise,
      lineDiscountPaise: t.lineDiscountPaise,

import { badRequest, conflict, notFound } from "../errors.js";
import { MOVEMENT_TYPES, StockMovement } from "../models/StockMovement.js";
import { Variant } from "../models/Variant.js";

// The ONLY way stock changes. Call it inside runInTransaction and pass the session:
// the stock update and its movement record are then saved together, or not at all.
//
// For a removal, the update itself says "only if enough stock is there". That makes
// it safe when many requests arrive at the same moment: they cannot oversell.
//
// `movementId` and `occurredAt` are used when a device syncs a change it made earlier:
// the movement keeps the device's id (so it can never be applied twice) and its time.
export async function applyStockChange({
  variantId,
  quantityChange,
  type,
  reason = "",
  referenceId = null,
  userId = null,
  allowNegative = false,
  movementId = null,
  occurredAt = null,
  session,
}) {
  if (!Number.isInteger(quantityChange) || quantityChange === 0) {
    throw badRequest(
      "Quantity change must be a non-zero whole number",
      "VALIDATION_ERROR",
    );
  }
  if (!MOVEMENT_TYPES.includes(type)) {
    throw badRequest(`Unknown movement type "${type}"`, "VALIDATION_ERROR");
  }

  const filter = { _id: variantId };
  if (quantityChange < 0 && !allowNegative) {
    filter.currentStock = { $gte: -quantityChange };
  }

  const variant = await Variant.findOneAndUpdate(
    filter,
    { $inc: { currentStock: quantityChange } },
    { new: true, session },
  );

  if (!variant) {
    const existing = await Variant.findById(variantId).session(session);
    if (!existing) throw notFound("Variant not found");
    throw conflict(
      "INSUFFICIENT_STOCK",
      `Only ${existing.currentStock} in stock for ${existing.sku}`,
      {
        variantId,
        available: existing.currentStock,
        requested: -quantityChange,
      },
    );
  }

  const [movement] = await StockMovement.create(
    [
      {
        ...(movementId ? { _id: movementId } : {}),
        variantId,
        type,
        quantityChange,
        stockAfter: variant.currentStock,
        reason,
        referenceId,
        userId,
        occurredAt: occurredAt ?? new Date(),
      },
    ],
    { session },
  );

  return { variant, movement };
}

import { runInTransaction } from "../db/transaction.js";
import {
  movementDataSchema,
  productDataSchema,
  variantDataSchema,
} from "../domain/syncSchemas.js";
import { saleCreateSchema } from "../domain/saleSchemas.js";
import { AppError } from "../errors.js";
import { translateDuplicate } from "../http/duplicates.js";
import { parse } from "../http/validate.js";
import { Product } from "../models/Product.js";
import { StockMovement } from "../models/StockMovement.js";
import { Variant } from "../models/Variant.js";
import { audit } from "./audit.js";
import { createSale } from "./sales.js";
import { applyStockChange } from "./stock.js";

const PRODUCT_FIELDS = ["name", "category", "brand", "supplier", "status"];
const VARIANT_FIELDS = [
  "sku",
  "barcode",
  "size",
  "costPaise",
  "mrpPaise",
  "sellingPricePaise",
  "taxRatePercent",
  "priceIncludesTax",
  "reorderThreshold",
  "status",
];

function requireAdmin(user) {
  if (user.role !== "admin") {
    throw new AppError(
      403,
      "FORBIDDEN",
      "Only an admin can sync catalogue and stock changes",
    );
  }
}

const isDuplicateId = (err) =>
  err?.code === 11000 &&
  Object.keys(err.keyPattern ?? err.keyValue ?? {}).includes("_id");

// Copies the listed fields onto a document and returns what actually changed.
function applyFields(doc, data, fields) {
  const changes = {};
  for (const field of fields) {
    if (doc[field] !== data[field]) {
      changes[field] = { from: doc[field], to: data[field] };
      doc[field] = data[field];
    }
  }
  return changes;
}

async function upsertProduct(data, req) {
  requireAdmin(req.user);

  // Tried twice: if two requests create the same product at once, the second
  // one finds it on its second try and simply updates it.
  for (let attempt = 0; attempt < 2; attempt += 1) {
    const existing = await Product.findById(data.id);

    if (!existing) {
      try {
        await Product.create({
          _id: data.id,
          ...Object.fromEntries(PRODUCT_FIELDS.map((f) => [f, data[f]])),
        });
        return { status: "applied", created: true, changed: true };
      } catch (err) {
        if (isDuplicateId(err)) continue;
        throw err;
      }
    }

    const changes = applyFields(existing, data, PRODUCT_FIELDS);
    if (Object.keys(changes).length === 0)
      return { status: "applied", created: false, changed: false };

    await existing.save();
    await audit(req, {
      action: "sync.product.updated",
      entity: "product",
      entityId: existing._id,
      details: { changes },
    });
    return { status: "applied", created: false, changed: true };
  }
  throw new AppError(
    409,
    "ID_EXISTS",
    "Could not save this product, please retry",
  );
}

async function upsertVariant(data, req) {
  requireAdmin(req.user);

  if (!(await Product.exists({ _id: data.productId }))) {
    throw new AppError(
      422,
      "PRODUCT_MISSING",
      "The product for this item is not on the server",
    );
  }

  for (let attempt = 0; attempt < 2; attempt += 1) {
    const existing = await Variant.findById(data.id);

    if (!existing) {
      try {
        await Variant.create({
          _id: data.id,
          productId: data.productId,
          currentStock: 0, // stock only ever arrives as movements
          ...Object.fromEntries(VARIANT_FIELDS.map((f) => [f, data[f]])),
        });
        return { status: "applied", created: true, changed: true };
      } catch (err) {
        if (isDuplicateId(err)) continue;
        throw translateDuplicate(err, data);
      }
    }

    if (existing.productId !== data.productId) {
      throw new AppError(
        409,
        "IMMUTABLE_FIELD",
        "An item cannot be moved to a different product",
      );
    }

    const changes = applyFields(existing, data, VARIANT_FIELDS);
    if (Object.keys(changes).length === 0)
      return { status: "applied", created: false, changed: false };

    try {
      await existing.save();
    } catch (err) {
      throw translateDuplicate(err, data);
    }
    await audit(req, {
      action: "sync.variant.updated",
      entity: "variant",
      entityId: existing._id,
      details: { sku: existing.sku, changes },
    });
    return { status: "applied", created: false, changed: true };
  }
  throw new AppError(
    409,
    "ID_EXISTS",
    "Could not save this item, please retry",
  );
}

async function applyMovement(data, req) {
  requireAdmin(req.user);

  if (await StockMovement.exists({ _id: data.id }))
    return { status: "duplicate" };

  try {
    const { variant } = await runInTransaction((session) =>
      applyStockChange({
        variantId: data.variantId,
        quantityChange: data.quantityChange,
        type: data.type,
        reason: data.reason,
        userId: req.user._id,
        // It already happened on the device, so it is recorded even if stock goes below zero.
        allowNegative: true,
        movementId: data.id,
        occurredAt: new Date(data.createdAt),
        session,
      }),
    );
    return { status: "applied", stockAfter: variant.currentStock };
  } catch (err) {
    if (isDuplicateId(err)) return { status: "duplicate" };
    throw err;
  }
}

async function applySale(data, req) {
  const parsed = parse(saleCreateSchema, data);
  // Whatever the device says, a synced sale already happened. It is saved and flagged,
  // never refused for stock or price rules.
  const { sale, duplicate } = await createSale({
    data: { ...parsed, offline: true },
    user: req.user,
  });

  if (!duplicate) {
    await audit(req, {
      action: "sale.created",
      entity: "sale",
      entityId: sale._id,
      details: {
        invoiceNumber: sale.invoiceNumber,
        totalPaise: sale.totalPaise,
        source: sale.source,
        flags: sale.flags.map((flag) => flag.code),
      },
    });
  }
  return {
    status: duplicate ? "duplicate" : "applied",
    invoiceNumber: sale.invoiceNumber,
    flags: sale.flags.map((flag) => flag.code),
  };
}

// Applies one operation. Throws an AppError for anything the server refuses.
export async function applyOperation(op, req) {
  switch (op.type) {
    case "product.upsert":
      return upsertProduct(parse(productDataSchema, op.data), req);
    case "variant.upsert":
      return upsertVariant(parse(variantDataSchema, op.data), req);
    case "stock.movement":
      return applyMovement(parse(movementDataSchema, op.data), req);
    case "sale.create":
      return applySale(op.data, req);
    default:
      throw new AppError(
        422,
        "UNSUPPORTED_TYPE",
        `Unknown operation type "${op.type}"`,
      );
  }
}
